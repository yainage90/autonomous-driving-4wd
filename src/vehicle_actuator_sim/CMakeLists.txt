cmake_minimum_required(VERSION 3.16)
project(vehicle_actuator_sim)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(vehicle_interfaces REQUIRED)
find_package(gz_sim_vendor REQUIRED)
find_package(gz_plugin_vendor REQUIRED)
find_package(gz-sim8 REQUIRED)
find_package(gz-plugin2 REQUIRED COMPONENTS register)
add_library(vehicle_actuator_watchdog SHARED src/WatchdogDiffDrive.cc)
target_include_directories(vehicle_actuator_watchdog PRIVATE include src)
ament_target_dependencies(vehicle_actuator_watchdog rclcpp vehicle_interfaces)
target_link_libraries(vehicle_actuator_watchdog gz-sim8::gz-sim8 gz-plugin2::register)
target_compile_options(vehicle_actuator_watchdog PRIVATE -Wall -Wextra)
install(TARGETS vehicle_actuator_watchdog LIBRARY DESTINATION lib)
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_watchdog test/test_watchdog.cpp)
  target_include_directories(test_watchdog PRIVATE include)
  ament_target_dependencies(test_watchdog vehicle_interfaces)
endif()
ament_package()
